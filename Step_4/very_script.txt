FILE = "my_fEMG_processed.acq"

CH_AK_BP  = 1      # AK band-pass output
CH_AK_NCH = 2      # AK band-pass + comb band-stop output
CH_AK_RMS = 3      # AK RMS envelope output

LOW, HIGH = 20.0, 450.0    # band-pass corners typed into AK
MAINS     = 50             # 50 or 60 Hz — must match AK
NOTCH_BW  = 2.0            # half-width of each stop band, Hz

MAX_LAG_S = 0.25           # search window for lag detection

def fir_zero_phase(sig, taps):
    """Symmetric FIR kernel via convolution, 'same' length, no phase shift."""
    return np.convolve(sig, taps, mode="same")

def find_lag(x, y, max_lag):
    """Integer sample shift of x relative to y that maximises correlation."""
    xa, ya = x - x.mean(), y - y.mean()
    c    = correlate(xa, ya, mode="full")
    lags = np.arange(-(len(xa) - 1), len(ya))
    keep = np.abs(lags) <= max_lag
    return int(lags[keep][np.argmax(c[keep])])

def compare(label, mine, theirs):
    k = int(EDGE_S * fs)
    x, y = mine[k:n - k], theirs[k:n - k]

    lag = find_lag(x, y, int(MAX_LAG_S * fs))

    # re-align before scoring amplitude agreement
    if lag > 0:
        xa, ya = x[lag:], y[:-lag]
    elif lag < 0:
        xa, ya = x[:lag], y[-lag:]
    else:
        xa, ya = x, y

    r     = np.corrcoef(xa, ya)[0, 1]
    r2    = r ** 2
    rmse  = np.sqrt(np.mean((xa - ya) ** 2))
    rng   = ya.max() - ya.min()
    nrmse = 100 * rmse / rng if rng else np.nan
    maxd  = np.abs(xa - ya).max()
    gain  = np.std(xa) / np.std(ya) if np.std(ya) else np.nan
    bias  = np.mean(xa - ya)

    ok = (r2 > R2_PASS and abs(lag) <= LAG_TOL
          and abs(gain - 1) < GAIN_TOL)
    verdict = "PASS" if ok else "FAIL"
